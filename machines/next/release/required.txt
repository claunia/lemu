# What qemu-system-m68k must have for the NeXT kits. check.sh fails if one
# is missing. WHERE is all or a platform (macos, linux).
#
# WHERE   KIND     NAME                       # why
all       version  git-id                     # --version names the commit
all       monitor  hmp                        # run.sh uses -monitor stdio
all       accel    tcg                        # the only accelerator for m68k
all       netdev   user                       # run.sh uses -nic user
macos     audio    coreaudio                  # sound on the Mac
linux     audio    pa                         # the default sound on Linux
linux     audio    alsa                       # sound on Linux hosts without PulseAudio
all       format   qcow2                      # the kit disks
all       format   raw                        # nvram.bin
all       machine  next-computer              # ns22computer
all       machine  next-station               # ns33station
all       machine  next-station-color         # os42color
all       machine  next-station-turbo-color   # ns33turbocolor
all       device   scsi-hd                    # the kit disks
